cmake_minimum_required(VERSION 3.20)
project(SeqProSequenceTextAddSubdirectoryConsumer LANGUAGES CXX)

if(NOT DEFINED SEQPRO_SOURCE_DIR OR NOT DEFINED SEQPRO_SEQUENCE_TEXT_DIR)
  message(FATAL_ERROR "SeqPro SequenceText source paths are required")
endif()

set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(SEQPRO_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(SEQPRO_BUILD_SEQUENCE_TEXT ON CACHE BOOL "" FORCE)
add_subdirectory(${SEQPRO_SOURCE_DIR} seqpro-build)

if(NOT TARGET SeqPro::seqpro OR NOT TARGET SeqPro::sequence_text)
  message(FATAL_ERROR "The SequenceText source-tree targets are unavailable")
endif()

add_executable(
  seqpro-sequence-text-consumer
  ${SEQPRO_SEQUENCE_TEXT_DIR}/tests/consumer/main.cc
)
target_link_libraries(seqpro-sequence-text-consumer PRIVATE SeqPro::sequence_text)
target_compile_features(seqpro-sequence-text-consumer PRIVATE cxx_std_17)

add_executable(
  seqpro-sequence-text-consumer-cxx20
  ${SEQPRO_SEQUENCE_TEXT_DIR}/tests/consumer/main.cc
)
target_link_libraries(seqpro-sequence-text-consumer-cxx20 PRIVATE SeqPro::sequence_text)
target_compile_features(seqpro-sequence-text-consumer-cxx20 PRIVATE cxx_std_20)
