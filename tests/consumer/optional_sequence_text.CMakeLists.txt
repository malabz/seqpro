cmake_minimum_required(VERSION 3.20)
project(SeqProOptionalSequenceTextConsumer LANGUAGES CXX)

find_package(SeqPro 0.2 CONFIG REQUIRED OPTIONAL_COMPONENTS SequenceText)

  message(FATAL_ERROR "The core target is missing when SequenceText is optional")
endif()
if(TARGET SeqPro::sequence_text OR SeqPro_SequenceText_FOUND)
  message(FATAL_ERROR "A core-only package unexpectedly provides SequenceText")
endif()
